### Детекция лиц и людей (MediaPipe)

```python
import mediapipe as mp

face_mesh = mp.solutions.face_detection.FaceDetection(
    model_selection=0, min_detection_confidence=0.5
)
pose = mp.solutions.pose.Pose(
    model_complexity=0, min_detection_confidence=0.5
)

def analyze_content(frame: np.ndarray) -> ContentMetrics:
    rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

    # Лица
    face_result = face_mesh.process(rgb)
    has_face = bool(face_result.detections)
    face_area = 0.0
    if has_face:
        bbox = face_result.detections[0].location_data.relative_bounding_box
        face_area = bbox.width * bbox.height

    # Тело
    pose_result = pose.process(rgb)
    has_person = bool(pose_result.pose_landmarks)

    # Saliency
    _, sal_map = cv2.saliency.StaticSaliencySpectralResidual_create().computeSaliency(frame)
    has_subject = float(sal_map.max()) > 0.5

    return ContentMetrics(
        has_face=has_face,
        face_area_ratio=face_area,
        has_person=has_person,
        has_subject=has_subject,
        scene_type=classify_shot_type(face_area, has_face, has_subject),
    )
```

### Определение типа плана

```python
def classify_shot_type(face_area: float, has_face: bool, has_subject: bool) -> str:
    if has_face:
        if face_area > 0.15:  return 'close'
        if face_area > 0.04:  return 'medium'
        return 'wide'
    if has_subject:           return 'medium'
    return 'wide'
```

### Детекция дубликатов (Perceptual Hash)

```python
import imagehash
from PIL import Image

def compute_phash(frame: np.ndarray) -> str:
    pil = Image.fromarray(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    return str(imagehash.phash(pil))

def are_duplicates(h1: str, h2: str, threshold: int = 8) -> bool:
    return (imagehash.hex_to_hash(h1) - imagehash.hex_to_hash(h2)) < threshold
```

Из группы дублей → оставить с наибольшим `quality_score`.
Остальные → помечать `is_duplicate=True` в таблице `fragments` (не удалять).

### Crop-пригодность 9:16

```python
def crop_suitability_9_16(frame: np.ndarray, saliency_map: np.ndarray) -> float:
    h, w = frame.shape[:2]
    target_w = int(h * 9 / 16)
    if target_w >= w:
        return 1.0
    x_start = (w - target_w) // 2
    x_end   = x_start + target_w
    total = saliency_map.sum()
    if total < 1e-6:
        return 0.5
    return float(saliency_map[:, x_start:x_end].sum() / total)
```

### Что это даёт пользователю

- Стилевые scores для F3/F5 стали значительно точнее (`social_score` теперь знает о лицах)
- Дубли не попадают в монтаж
- Детекция «закрытых глаз» или «размытых лиц» возможна на этом же уровне

### Видео-превью (3-секундный клип)

```python
def generate_video_preview(
    source_path: str,
    start_s: float,
    end_s: float,
    output_path: str,
    max_duration: float = 3.0,
) -> str:
    mid = (start_s + end_s) / 2.0
    preview_start = max(start_s, mid - max_duration / 2)
    preview_end   = min(end_s,   preview_start + max_duration)

    cmd = [
        'ffmpeg', '-y',
        '-ss', str(preview_start),
        '-i', source_path,
        '-t', str(preview_end - preview_start),
        '-vf', 'scale=320:180,fps=24',
        '-c:v', 'libx264', '-preset', 'ultrafast',
        '-an',
        output_path,
    ]
    subprocess.run(cmd, check=True, capture_output=True)
    return output_path
```

## Уровень 4 — Семантические теги (CLIP)

**Срок:** +2–3 недели. Требует GPU или Apple Silicon MPS.

### CLIP-теги сцены

```python
import open_clip
import torch

model, _, preprocess = open_clip.create_model_and_transforms(
    'ViT-B-32', pretrained='openai'
)
tokenizer = open_clip.get_tokenizer('ViT-B-32')

SCENE_TAGS = ['nature', 'city', 'interior', 'ocean', 'mountains',
              'people', 'sport', 'food', 'architecture', 'abstract']

def classify_scene(frame: np.ndarray) -> str:
    pil = Image.fromarray(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    image = preprocess(pil).unsqueeze(0)
    texts = tokenizer([f"a photo of {t}" for t in SCENE_TAGS])

    with torch.no_grad():
        image_features = model.encode_image(image)
        text_features  = model.encode_text(texts)
        probs = (image_features @ text_features.T).softmax(dim=-1)

    return SCENE_TAGS[probs.argmax().item()]
```

### Что это даёт

- Фильтрация «природа vs город» без ручной разметки
- `travel_score` для F6 теперь включает тег `nature`/`ocean`/`mountains`
- В UI: фильтр по тегу сцены (`🌅 природа`, `🏙 город`, `🏠 интерьер`)
