```python
from src.storage.analysis_db import AnalysisDB

    db.save_source_file('/tmp/test.mp4')

def test_cache_hit_after_analyze(db, tmp_path):
    # Создаём реальный файл, сохраняем, помечаем analyzed
    f = tmp_path / 'test.mp4'
    f.write_bytes(b'\x00' * 1024)
    sid = db.save_source_file(str(f))

def test_cache_invalid_after_modification(db, tmp_path):
    f = tmp_path / 'test.mp4'
    f.write_bytes(b'\x00' * 1024)
    db.save_source_file(str(f))
    db.mark_analyzed(str(f))
    # Изменяем файл
    f.write_bytes(b'\xFF' * 2048)
    assert db.is_analyzed(str(f)) is False  # хэш изменился

    f = tmp_path / 'test.mp4'
    f.write_bytes(b'\x00' * 1024)
    sid = db.save_source_file(str(f))
    fragments = [

        {'start_s': 5.0, 'end_s': 9.0, 'duration_s': 4.0, 'quality_score': 0.82},
    ]
    db.save_fragments(sid, fragments)
```

### test_quality_analyzer.py

```python
import pytest
import numpy as np
import cv2
from src.preprocessing.quality_analyzer import QualityAnalyzer

analyzer = QualityAnalyzer()

def test_black_frame_detection():
    black = np.zeros((720, 1280, 3), dtype=np.uint8)
    m = analyzer._analyze_frame(black)
    assert m.is_black is True
    assert m.brightness < 0.05

def test_sharp_frame():
    # Шахматный паттерн — высокая резкость
    frame = np.zeros((200, 200, 3), dtype=np.uint8)
    frame[::10, :] = 255
    m = analyzer._analyze_frame(frame)
    assert m.sharpness > 0.40

def test_blurry_frame():
    frame = np.full((200, 200, 3), 128, dtype=np.uint8)
    blurred = cv2.GaussianBlur(frame, (51, 51), 20)
    m = analyzer._analyze_frame(blurred)
    assert m.sharpness < 0.05

def test_brightness_ranges():
    bright = np.full((100, 100, 3), 250, dtype=np.uint8)
    dark   = np.full((100, 100, 3), 10,  dtype=np.uint8)
    assert analyzer._analyze_frame(bright).brightness > 0.90
    assert analyzer._analyze_frame(dark).brightness   < 0.10

def test_overexpose_detection():
    overexposed = np.full((100, 100, 3), 253, dtype=np.uint8)
    m = analyzer._analyze_frame(overexposed)
    assert m.overexpose is True
```

### test_clip_scorer.py

```python
from src.preprocessing.quality_analyzer import TechMetrics
from src.preprocessing.clip_scorer import ClipScorer

scorer = ClipScorer()

def make_metrics(**kwargs):
    defaults = dict(
        sharpness=0.7, brightness=0.5, stability=0.8,
        motion=0.2, action=0.2, calm=0.7,
        is_black=False, is_corrupt=False, overexpose=False,
    )
    defaults.update(kwargs)
    return TechMetrics(**defaults)

def test_good_clip_high_quality():
    m = make_metrics(sharpness=0.9, brightness=0.6, stability=0.9, motion=0.15)
    scores = scorer.score(m)
    assert scores.quality_score > 0.70

def test_black_clip_low_quality():
    m = make_metrics(is_black=True, brightness=0.02, sharpness=0.01)
    scores = scorer.score(m)
    assert scores.quality_score < 0.20

def test_cinematic_vs_action():
    stable = make_metrics(stability=0.9, motion=0.1)
    dynamic = make_metrics(stability=0.3, motion=0.8, action=0.7)
    s_stable  = scorer.score(stable)
    s_dynamic = scorer.score(dynamic)
    assert s_stable.cinematic_score > s_dynamic.cinematic_score
    assert s_dynamic.action_score   > s_stable.action_score

def test_quality_score_bounds():
    m = make_metrics()
    scores = scorer.score(m)
    assert 0.0 <= scores.quality_score <= 1.0
    assert 0.0 <= scores.cinematic_score <= 1.0
    assert 0.0 <= scores.action_score <= 1.0
```

### test_pipeline_integration.py

```python
import time, os
import pytest
from src.preprocessing.pipeline import analyze_project
from src.storage.analysis_db import AnalysisDB
from src.preprocessing.clip_library import FragmentLibrary

TEST_PROJECT = os.environ.get('TEST_PROJECT_DIR', 'tests/fixtures/project')

@pytest.mark.integration
def test_full_pipeline_runs():
    n = analyze_project(TEST_PROJECT)
    assert n >= 5, f"Expected >= 5 fragments, got {n}"

@pytest.mark.integration
def test_cache_works():
    analyze_project(TEST_PROJECT)  # первый запуск (прогревает кэш)
    start = time.time()
    analyze_project(TEST_PROJECT)  # второй запуск (из кэша)
    assert time.time() - start < 2.0, "Cache miss — повторный анализ слишком долгий"

@pytest.mark.integration
def test_fragment_quality_threshold():
    analyze_project(TEST_PROJECT)
    db  = AnalysisDB(TEST_PROJECT)
    lib = FragmentLibrary(db)
    candidates = lib.get_candidates(TEST_PROJECT, min_quality=0.55)
    for c in candidates:
        assert c.features.technical_quality_score >= 0.50  # допуск 0.05

@pytest.mark.integration
def test_no_black_frames_in_candidates():
    analyze_project(TEST_PROJECT)
    db  = AnalysisDB(TEST_PROJECT)
    lib = FragmentLibrary(db)
    candidates = lib.get_candidates(TEST_PROJECT, min_quality=0.08)
    # Кандидаты не должны иметь очень низкий quality_score
    for c in candidates:
        assert c.features.brightness_score >= 0.08

@pytest.mark.integration
def test_thumbnails_generated():
    analyze_project(TEST_PROJECT)
    db  = AnalysisDB(TEST_PROJECT)
    lib = FragmentLibrary(db)
    fragments = lib.get_fragments_for_ui(TEST_PROJECT, limit=10)

        assert f.thumbnail_path is not None
        assert os.path.exists(f.thumbnail_path), f"Missing: {f.thumbnail_path}"
```

## 3. Ручные качественные тесты

### Тест Q-01: Отсев брака

**Материал:** набор видео с намеренным браком:
- `dark_clip.mp4` — снято в темноте (brightness ~ 0.05)
- `shaky_clip.mp4` — сильная тряска на ходьбе
- `blurry_clip.mp4` — вся запись в расфокусе
- `good_clip.mp4` — нормальное качество

**Ожидание:**
- `dark_clip.mp4` → 0 фрагментов с quality_score > 0.08
- `shaky_clip.mp4` → 0 фрагментов со stability > 0.20
- `blurry_clip.mp4` → 0 фрагментов с sharpness > 0.08
- `good_clip.mp4` → 5+ фрагментов с quality_score > 0.55

### Тест Q-02: Кэш при изменении папки

**Шаги:**
1. Запустить анализ на папке с 5 видео → замерить время T1
2. Добавить 1 новое видео в папку
3. Запустить анализ снова → замерить время T2

**Ожидание:**
- T2 ≈ T1 / 5 (анализируется только 1 новый файл)
- Старые фрагменты сохраняются без изменений

### Тест Q-03: Визуальное качество thumbnails

**Шаги:**
1. Запустить анализ на реальном проекте
2. Открыть страницу «Фрагменты» в GUI

**Ожидание:**
- Все thumbnails читаемы (не чёрные, не размытые в превью)
- Временной диапазон в карточке соответствует contentu thumbnails
- Рейтинг звёздочками коррелирует с визуальным качеством

### Тест Q-04: Конечный результат рендера

**Шаги:**
1. Проект с 15+ видео (смесь хороших и плохих)
2. Рендер без анализа → смотреть результат
3. Рендер с анализом (min_quality=0.55) → смотреть результат

**Ожидание:**
- В ролике с анализом нет чёрных/смазанных/трясущихся кадров
- Субъективное качество ролика с анализом выше

### Тест Q-05: Производительность на реальных данных

**Материал:** папка с 10 видео, суммарно 30–60 мин.

| Метрика | Ожидание |
|---|---|
| Время первого анализа | < 3 мин |
| Время повторного анализа | < 2 с |
| Количество фрагментов | 50–200 |
| CPU-нагрузка в процессе | < 80% на 1 ядро (параллелизм работает) |
| RAM потребление | < 500 MB |

### Тест Q-06: Интеграция со стилями

**Шаги:**
1. Анализировать папку с action-видео (спорт, велосипед, скейт)

**Ожидание:**
