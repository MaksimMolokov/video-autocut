## 1. Точка интеграции в gui.py

Существующий пайплайн при рендере в gui.py (упрощённо):

```python

    video_path, good_ranges, min_duration, max_duration

```

После добавления предобработки:

```python
# ПОСЛЕ — добавить ветку с чтением из БД
from src.storage.analysis_db import AnalysisDB
from src.preprocessing.clip_library import FragmentLibrary

db  = AnalysisDB(project_dir)
lib = FragmentLibrary(db)

if lib.has_analyzed(video_path):
    # Читаем кандидатов из кэша предобработки
    candidates = lib.get_candidates(
        source_path=video_path,
        style_score_key=f'{preset_id}_score',  # 'cinematic_score', 'action_score', ...
        min_quality=preset_settings.min_quality_score,
        approved_only=(preprocessing_mode != 'auto'),
    )
    logger.info(f"[Preprocessing] {len(candidates)} candidates from cache")
else:
    # Fallback: стандартный путь без предобработки
    good_ranges = analyzer.get_good_ranges(video_path, preset_id)
    candidates  = CandidateClipBuilder.build_candidates(
        video_path, good_ranges, min_duration, max_duration
    )
    logger.info("[Preprocessing] No cache — using live analysis")
```

**Ключевое свойство:** существующий пайплайн не ломается.
Предобработка — это просто более качественный источник `candidates`.

## 2. API класса FragmentLibrary

```python
# src/preprocessing/clip_library.py

class FragmentLibrary:

    def has_analyzed(self, video_path: str) -> bool:
        """Есть ли в кэше анализ для этого файла (и он актуален)."""

    def get_candidates(
        self,
        source_path: str,
        style_score_key: str = 'quality_score',
        min_quality: float   = 0.50,
        min_duration: float  = 1.0,
        max_duration: float  = 10.0,
        approved_only: bool  = False,
        limit: int           = 200,
    ) -> List[CandidateClip]:
        """
        Вернуть CandidateClip-объекты из БД.
        Формат совместим с существующим CandidateClipBuilder.
        """

    def get_all_candidates_for_project(
        self,
        project_dir: str,
        style_score_key: str = 'quality_score',
        **kwargs,
    ) -> List[CandidateClip]:
        """Все кандидаты из всех проанализированных файлов проекта."""

    def mark_approved(self, fragment_id: int, approved: bool):
        """Одобрить или отключить фрагмент (user_approved = 1 / 0)."""

    def mark_priority(self, fragment_id: int, priority: bool):
        """Пометить фрагмент как избранный (user_priority = 1 / NULL)."""

    def get_fragments_for_ui(
        self, project_dir: str,
        min_quality: float = 0.50,
        sort_by: str = 'quality_score',
        limit: int = 200,
    ) -> List[FragmentData]:
        """Для UI: с thumbnail_path, human-readable полями."""
```

### Возвращаемый CandidateClip

`FragmentLibrary.get_candidates()` должен возвращать объекты, совместимые
с уже существующим `CandidateClip` из `src/video_analysis/candidate_builder.py`.
Это достигается путём конвертации:

```python
def _fragment_to_candidate(f: FragmentRow) -> CandidateClip:
    features = ClipFeatures(
        start_time             = f.start_s,
        end_time               = f.end_s,
        duration               = f.duration_s,
        sharpness_score        = f.sharpness or 0.5,
        brightness_score       = f.brightness or 0.5,
        contrast_score         = f.contrast or 0.5,
        motion_score           = f.motion or 0.5,
        camera_stability_score = f.stability or 0.5,
        action_score           = f.action or 0.5,
        calm_score             = f.calm or 0.5,
        technical_quality_score= f.quality_score or 0.5,
    )
    candidate = CandidateClip(
        source_path  = f.source_path,
        start        = f.start_s,
        end          = f.end_s,
        duration     = f.duration_s,
        features     = features,
        is_must_use  = bool(f.user_priority),
        final_score  = f.quality_score or 0.5,
    )
    return candidate
```

## 3. Параллельная обработка проекта

```python
# src/preprocessing/pipeline.py

from concurrent.futures import ProcessPoolExecutor, as_completed
import os

def analyze_project(
    project_dir: str,
    progress_callback: Callable[[str, int, int], None] = None,
    force_reanalyze: bool = False,
) -> int:
    """
    Запустить предобработку для всех видео в project_dir.

    Returns: количество найденных фрагментов
    """
    db       = AnalysisDB(project_dir)
    videos   = list_video_files(project_dir)   # .mp4, .mov, .avi, ...

    if force_reanalyze:
        db.reset_all()

    pending = [v for v in videos if not db.is_analyzed(v)]

    if not pending:
        return db.count_fragments(project_dir)

    cpu_count = max(1, os.cpu_count() // 2)

    with ProcessPoolExecutor(max_workers=cpu_count) as executor:
        futures = {
            executor.submit(_analyze_one_video, str(v), str(project_dir)): v
            for v in pending
        }
        completed = 0
        for future in as_completed(futures):
            video_path = futures[future]
            try:
                fragments = future.result()
                db.save_fragments(video_path, fragments)
                completed += 1
                if progress_callback:
                    progress_callback(
                        str(video_path.name), completed, len(pending)
                    )
            except Exception as e:
                logger.error(f"Analysis failed for {video_path}: {e}")
                db.mark_error(str(video_path), str(e))

    return db.count_fragments(project_dir)


def _analyze_one_video(video_path: str, project_dir: str) -> List[FragmentData]:
    """
    Запускается в отдельном процессе.
    Импорты внутри функции — ProcessPoolExecutor требует picklable.
    """
    from src.preprocessing.scene_detector  import SceneDetector
    from src.preprocessing.quality_analyzer import QualityAnalyzer
    from src.preprocessing.clip_scorer      import ClipScorer

    scenes    = SceneDetector().split(video_path, min_scene_length=0.8)
    fragments = []

    for scene in scenes:
        metrics   = QualityAnalyzer().analyze(video_path, scene)
        if metrics.is_corrupt or metrics.is_black:
            continue    # жёсткий отсев брака
        scores    = ClipScorer().score(metrics)
        fragments.append(FragmentData(
            start_s=scene.start_s, end_s=scene.end_s,
            **metrics.__dict__, **scores,
        ))

    return fragments
```

## 4. Интеграция с TimelineBuilder

`TimelineBuilder` получает `candidates: List[CandidateClip]`.
После добавления предобработки он может получать их из двух источников:

1. **С предобработкой:** `FragmentLibrary.get_all_candidates_for_project()`
2. **Без предобработки:** существующий `CandidateClipBuilder.build_candidates()`

Никаких изменений в `TimelineBuilder` не нужно. Он просто получает
более качественный пул кандидатов.

## 5. Производительность

| Этап | Время на 10 мин видео | Параллелизм |
|---|---|---|
| Разбивка на сцены (PySceneDetect) | ~15–25 с | по видео |
| Анализ метрик (OpenCV, выборка 1/15 кадр) | ~8–15 с | по сценам |
| Thumbnail генерация (FFmpeg) | ~3–5 с | по фрагментам |
| Контентный анализ (MediaPipe, уровень 2) | +20–40 с | по ключевым кадрам |
| **Итого MVP** | **~25–45 с** | CPU cores / 2 |
| **Итого с контентом** | **~50–90 с** | CPU cores / 2 |

Результаты кэшируются → повторный запуск < 1 с.
