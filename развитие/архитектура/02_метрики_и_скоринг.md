# Метрики и алгоритмы скоринга фрагментов

## 1. Технические метрики (уровень MVP)

Все метрики нормализованы в [0.0, 1.0].

### Резкость

```python
# Дисперсия Лапласиана — чем выше, тем резче
gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
laplacian_var = cv2.Laplacian(gray, cv2.CV_64F).var()
sharpness = min(1.0, laplacian_var / 500.0)

# Порог отсева: sharpness < 0.08 → брак
```

### Яркость

```python
hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
brightness = float(hsv[:, :, 2].mean()) / 255.0

# Порог отсева:
# brightness < 0.12 → слишком тёмный
# brightness > 0.92 → пересвет
```

### Пересвет

```python
overexposed_ratio = float((frame > 250).mean())
# overexposed_ratio > 0.15 → флаг пересвета
```

### Тряска камеры (нестабильность)

```python
# Variance of optical flow vectors — чем выше, тем сильнее тряска
flow = cv2.calcOpticalFlowFarneback(prev_gray, curr_gray, ...)
flow_mag = np.sqrt(flow[..., 0]**2 + flow[..., 1]**2)
instability = float(flow_mag.std())       # высокое std = хаотичное движение
stability = max(0.0, 1.0 - instability / 10.0)

# Порог отсева: stability < 0.20 → слишком трясёт
```

### Скорость движения

```python
motion = float(flow_mag.mean()) / 10.0   # нормализация
# 0.0–0.05 = статичный кадр
# 0.05–0.50 = плавное движение
# > 0.70 = очень быстрое или хаотичное
```

### Чёрный кадр / технический брак

```python
is_black = brightness < 0.04
is_white = brightness > 0.96
is_corrupt = sharpness < 0.02 and brightness < 0.10
```

### Пустота кадра

```python
edges = cv2.Canny(gray, 50, 150)
edge_density = float(edges.mean()) / 255.0
is_empty = edge_density < 0.02 and brightness < 0.90
```

## 2. Технический скоринг фрагмента

Анализируется по нескольким равномерно распределённым ключевым кадрам
(обычно 5–10 кадров на сцену). Метрики усредняются.

```python
def compute_technical_metrics(frames: List[np.ndarray]) -> TechMetrics:
    """
    frames: список ключевых кадров сцены (BGR numpy arrays)
    """
    sharpnesses  = [laplacian_sharpness(f) for f in frames]
    brightnesses = [mean_brightness(f) for f in frames]
    stabilities  = [camera_stability(frames[i], frames[i+1])
                    for i in range(len(frames)-1)]
    motions      = [optical_flow_magnitude(frames[i], frames[i+1])
                    for i in range(len(frames)-1)]

    return TechMetrics(
        sharpness  = float(np.median(sharpnesses)),
        brightness = float(np.mean(brightnesses)),
        stability  = float(np.mean(stabilities)),
        motion     = float(np.mean(motions)),
        # Производные
        action     = motion * (1.0 - stability * 0.5),
        calm       = stability * (1.0 - motion * 0.5),
        # Брак-флаги
        is_black   = all(b < 0.05 for b in brightnesses),
        is_corrupt = np.median(sharpnesses) < 0.02,
        overexpose = any((f > 250).mean() > 0.15 for f in frames),
    )
```

## 3. Контентные метрики (уровень 2)

### Детекция лиц и людей

```python
import mediapipe as mp

face_detection = mp.solutions.face_detection.FaceDetection(
    model_selection=0, min_detection_confidence=0.5
)

def detect_faces(frame: np.ndarray) -> dict:
    rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
    results = face_detection.process(rgb)
    if not results.detections:
        return {'has_face': False, 'face_area_ratio': 0.0}
    max_bbox = max(results.detections,
                   key=lambda d: d.location_data.relative_bounding_box.width)
    bbox = max_bbox.location_data.relative_bounding_box
    area = bbox.width * bbox.height
    return {
        'has_face': True,
        'face_area_ratio': float(area),
        'face_count': len(results.detections),
    }
```

### Тип плана (heuristic)

```python
def classify_shot_type(face_area_ratio: float, has_face: bool,
                       has_subject: bool) -> str:
    if has_face:
        if face_area_ratio > 0.15:
            return 'close'     # крупный план: лицо занимает > 15% кадра
        elif face_area_ratio > 0.04:
            return 'medium'    # средний план
        else:
            return 'wide'      # широкий: человек виден мелко
    elif has_subject:
        return 'medium'
    else:
        return 'wide'          # пейзаж, общий план
```

_, saliency_map = saliency.computeSaliency(frame)

### Пригодность для crop 9:16

```python
def crop_suitability_9_16(frame: np.ndarray) -> float:
    """
    Проверяем, не теряется ли важный объект при центральном кропе
    для вертикального формата.
    """
    h, w = frame.shape[:2]
    target_w = int(h * 9 / 16)
    if target_w >= w:
        return 1.0  # уже вертикальный

    # Центральная полоса шириной target_w
    x_start = (w - target_w) // 2
    x_end   = x_start + target_w

    _, saliency_map = saliency.computeSaliency(frame)
    total_sal = saliency_map.sum()
    if total_sal < 1e-6:
        return 0.5

    crop_sal  = saliency_map[:, x_start:x_end].sum()
    return float(crop_sal / total_sal)  # доля важности в центральной зоне
```

## 4. Агрегированный quality_score

```python
def compute_quality_score(m: TechMetrics, c: ContentMetrics = None) -> float:
    brightness_ok = 1.0 if 0.20 < m.brightness < 0.88 else 0.3
    motion_ok     = 1.0 if 0.04 < m.motion < 0.72 else 0.4
    overexpose_ok = 1.0 - (0.7 if m.overexpose else 0.0)

    base = (
        0.25 * m.sharpness    +
        0.20 * brightness_ok  +
        0.20 * m.stability    +
        0.15 * overexpose_ok  +
        0.10 * motion_ok      +
        0.10 * (c.has_subject if c else 0.5)
    )
    return round(min(1.0, max(0.0, base)), 3)
```

### Стилевые оценки

```python
def compute_style_scores(m: TechMetrics, c: ContentMetrics = None) -> dict:
    return {
        # F2: cinematic — стабильность и резкость важнее всего
        'cinematic_score': (
            0.35 * m.stability  +
            0.30 * m.sharpness  +
            0.20 * (1 - min(1, m.motion * 1.5)) +  # плавное движение, не хаос
            0.15 * m.brightness
        ),
        # F4: action — движение и динамика
        'action_score': (
            0.40 * m.motion     +
            0.30 * m.action     +
            0.20 * m.sharpness  +
            0.10 * (1 - m.stability)  # немного нестабильности — ок для спорта
        ),
        # F3: social — движение + люди
        'social_score': (
            0.25 * m.motion     +
            0.25 * (c.has_face if c else 0.0) +
            0.25 * m.sharpness  +
            0.25 * m.brightness
        ),
        # F5: premium — максимальное качество и спокойствие
        'premium_score': (
            0.40 * m.stability  +
            0.35 * m.sharpness  +
            0.25 * m.brightness
        ),
        # F6: travel — цвет, атмосфера, разнообразие
        'travel_score': (
            0.30 * m.sharpness  +
            0.25 * m.brightness +
            0.25 * m.stability  +
            0.20 * (c.has_face if c else 0.0)
        ),
    }
```

## 5. Обнаружение дубликатов

### Perceptual hash (imagehash)

```python
import imagehash
from PIL import Image

def compute_phash(frame: np.ndarray) -> str:
    pil = Image.fromarray(cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
    return str(imagehash.phash(pil))

def are_duplicates(hash1: str, hash2: str, threshold: int = 8) -> bool:
    h1 = imagehash.hex_to_hash(hash1)
    h2 = imagehash.hex_to_hash(hash2)
    return (h1 - h2) < threshold  # расстояние Хэмминга
```

### Правило дублей

Из группы фрагментов с расстоянием Хэмминга < 8 оставлять только один —
с наибольшим `quality_score`. Остальные помечать в таблице `duplicates`
(не удалять — чтобы пользователь мог их увидеть и вернуть вручную).

### Временные дубли (из одного файла)

```python
def is_temporal_duplicate(f1: Fragment, f2: Fragment,
                           overlap_threshold: float = 0.7) -> bool:
    if f1.source_id != f2.source_id:
        return False
    overlap_start = max(f1.start_s, f2.start_s)
    overlap_end   = min(f1.end_s,   f2.end_s)
    if overlap_end <= overlap_start:
        return False
    overlap = overlap_end - overlap_start
    min_dur = min(f1.duration_s, f2.duration_s)
    return (overlap / min_dur) > overlap_threshold
```

## 6. Критерии автоматического отсева

Фрагмент автоматически исключается (не попадает в candidates), если:

| Условие | Описание |
|---|---|
| `sharpness < 0.08` | Полностью смазанный кадр |
| `brightness < 0.10` | Чёрный или очень тёмный |
| `brightness > 0.94` | Пересвет |
| `stability < 0.15` | Сильная тряска |
| `duration_s < 0.8` | Слишком короткий |
| `is_black = True` | Технический чёрный кадр |
| `is_corrupt = True` | Encoding artifact |
| `is_duplicate = True` | Дубль (сохраняется в duplicates, не в candidates) |
| `edge_density < 0.015` | Пустой, бессодержательный кадр |

Пороги настраиваются в `src/preprocessing/quality_analyzer.py` и могут
адаптивно снижаться, если после отсева остаётся < 15 фрагментов.
